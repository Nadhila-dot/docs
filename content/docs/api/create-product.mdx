```go
package main

import (
    "fmt"
    "io"
    "net/http"
    "os"
    "time"
    "strings"
)

func main() {
    token := os.Getenv("CTRLPANEL_API_TOKEN")
    if token == "" { panic("Set CTRLPANEL_API_TOKEN first") }
    request, err := http.NewRequest("POST", "https://ctrlpanel.hosting.gg/api/products", strings.NewReader("{\"name\":\"Medium Minecraft\",\"description\":\"Perfect for your SMP with friends\",\"price\":10,\"minimum_credits\":10,\"memory\":4096,\"cpu\":200,\"swap\":0,\"disk\":20480,\"io\":500,\"serverlimit\":1,\"databases\":3,\"backups\":2,\"allocations\":1,\"nodes\":[1,2],\"eggs\":[1,3],\"disabled\":false,\"oom_killer\":true,\"default_billing_priority\":0,\"billing_period\":\"monthly\"}"))
    if err != nil { panic(err) }
    request.Header.Set("Authorization", "Bearer " + token)
    request.Header.Set("Accept", "application/json")
    request.Header.Set("Content-Type", "application/json")
    client := &http.Client{Timeout: 30 * time.Second}
    response, err := client.Do(request)
    if err != nil { panic(err) }
    defer response.Body.Close()
    body, err := io.ReadAll(response.Body)
    if err != nil { panic(err) }
    if response.StatusCode < 200 || response.StatusCode >= 300 {
        panic(fmt.Sprintf("HTTP %d: %s", response.StatusCode, body))
    }
    fmt.Println(string(body))
}
```

```rust
// Dependencies:
// cargo add reqwest
// cargo add tokio --features macros,rt-multi-thread
use std::{env, time::Duration};

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    let token = env::var("CTRLPANEL_API_TOKEN")?;
    let client = reqwest::Client::builder()
        .timeout(Duration::from_secs(30))
        .build()?;
    let response = client
        .request(reqwest::Method::POST, "https://ctrlpanel.hosting.gg/api/products")
        .bearer_auth(token)
        .header("Accept", "application/json")
        .header("Content-Type", "application/json")
        .body("{\"name\":\"Medium Minecraft\",\"description\":\"Perfect for your SMP with friends\",\"price\":10,\"minimum_credits\":10,\"memory\":4096,\"cpu\":200,\"swap\":0,\"disk\":20480,\"io\":500,\"serverlimit\":1,\"databases\":3,\"backups\":2,\"allocations\":1,\"nodes\":[1,2],\"eggs\":[1,3],\"disabled\":false,\"oom_killer\":true,\"default_billing_priority\":0,\"billing_period\":\"monthly\"}")
        .send()
        .await?;
    let status = response.status();
    let text = response.text().await?;
    if !status.is_success() {
        return Err(format!("HTTP {}: {}", status, text).into());
    }
    println!("{}", text);
    Ok(())
}
```

### Request body (application/json)

| Property | Type | Required | Description |
| --- | --- | --- | --- |
| `name` | string | Yes | Max 30 chars. |
| `description` | string | Yes | Max 191 chars. |
| `price` | number | Yes | Min 0. |
| `minimum_credits` | number | No | Min 0, must be GTE price. |
| `memory` | integer | Yes | Memory in MB. Use 0 for unlimited. |
| `cpu` | integer | Yes | CPU limit in %. Use 0 for unlimited. |
| `swap` | integer | Yes | Swap in MB. Use 0 to disable, -1 for unlimited. |
| `disk` | integer | Yes | Disk in MB. Use 0 for unlimited. |
| `io` | integer | Yes | IO limit. |
| `serverlimit` | integer | Yes | Max servers. Use 0 for unlimited. |
| `databases` | integer | Yes | Max databases. |
| `backups` | integer | Yes | Max backups. |
| `allocations` | integer | Yes | Max allocations. |
| `nodes` | array | No | List of node IDs associated with the product. |
| `eggs` | array | No | List of egg IDs associated with the product. |
| `disabled` | boolean | No | Whether the product is disabled. |
| `oom_killer` | boolean | No | Whether the OOM killer is enabled. |
| `default_billing_priority` | integer | No | The default billing priority. |
| `billing_period` | string | Yes | hourly, daily, weekly, monthly, quarterly, half-annually, annually. |
