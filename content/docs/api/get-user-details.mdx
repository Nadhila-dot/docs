<Tab value="go">

```go
package main

import (
    "fmt"
    "io"
    "net/http"
    "os"
    "time"
)

func main() {
    token := os.Getenv("CTRLPANEL_API_TOKEN")
    if token == "" { panic("Set CTRLPANEL_API_TOKEN first") }
    request, err := http.NewRequest("GET", "https://ctrlpanel.hosting.gg/api/users/1", nil)
    if err != nil { panic(err) }
    request.Header.Set("Authorization", "Bearer " + token)
    request.Header.Set("Accept", "application/json")
    client := &http.Client{Timeout: 30 * time.Second}
    response, err := client.Do(request)
    if err != nil { panic(err) }
    defer response.Body.Close()
    body, err := io.ReadAll(response.Body)
    if err != nil { panic(err) }
    if response.StatusCode < 200 || response.StatusCode >= 300 {
        panic(fmt.Sprintf("HTTP %d: %s", response.StatusCode, body))
    }
    fmt.Println(string(body))
}
```

<Tab value="rust">

```rust
// Dependencies:
// cargo add reqwest
// cargo add tokio --features macros,rt-multi-thread
use std::{env, time::Duration};

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    let token = env::var("CTRLPANEL_API_TOKEN")?;
    let client = reqwest::Client::builder()
        .timeout(Duration::from_secs(30))
        .build()?;
    let response = client
        .request(reqwest::Method::GET, "https://ctrlpanel.hosting.gg/api/users/1")
        .bearer_auth(token)
        .header("Accept", "application/json")
        .send()
        .await?;
    let status = response.status();
    let text = response.text().await?;
    if !status.is_success() {
        return Err(format!("HTTP {}: {}", status, text).into());
    }
    println!("{}", text);
    Ok(())
}
```

### Parameters

| Name | Location | Required | Description |
| --- | --- | --- | --- |
| `id` | path | Yes | The ID of the user. |
| `include` | query | No | Comma-separated list of related resources to include. <br />Available: <code>servers.product</code>, <code>notifications</code>, <code>payments</code>, <code>vouchers.users</code>, <code>roles.permissions</code>, <code>discordUser</code> |
| `filter[name]` | query | No | Filter results by name |
| `filter[server_limit]` | query | No | Filter results by server_limit |
| `filter[email]` | query | No | Filter results by email |
| `filter[pterodactyl_id]` | query | No | Filter results by pterodactyl_id |
| `filter[suspended]` | query | No | Filter results by suspended |
