<Tab value="go">

```go
package main

import (
    "fmt"
    "io"
    "net/http"
    "os"
    "time"
    "strings"
)

func main() {
    token := os.Getenv("CTRLPANEL_API_TOKEN")
    if token == "" { panic("Set CTRLPANEL_API_TOKEN first") }
    request, err := http.NewRequest("POST", "https://ctrlpanel.hosting.gg/api/vouchers", strings.NewReader("{\"memo\":\"Summer 2026 Promotion\",\"code\":\"SUMMER2026\",\"uses\":100,\"credits\":50,\"expires_at\":\"31-12-2026 23:59:59\"}"))
    if err != nil { panic(err) }
    request.Header.Set("Authorization", "Bearer " + token)
    request.Header.Set("Accept", "application/json")
    request.Header.Set("Content-Type", "application/json")
    client := &http.Client{Timeout: 30 * time.Second}
    response, err := client.Do(request)
    if err != nil { panic(err) }
    defer response.Body.Close()
    body, err := io.ReadAll(response.Body)
    if err != nil { panic(err) }
    if response.StatusCode < 200 || response.StatusCode >= 300 {
        panic(fmt.Sprintf("HTTP %d: %s", response.StatusCode, body))
    }
    fmt.Println(string(body))
}
```

```rust
// Dependencies:
// cargo add reqwest
// cargo add tokio --features macros,rt-multi-thread
use std::{env, time::Duration};

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    let token = env::var("CTRLPANEL_API_TOKEN")?;
    let client = reqwest::Client::builder()
        .timeout(Duration::from_secs(30))
        .build()?;
    let response = client
        .request(reqwest::Method::POST, "https://ctrlpanel.hosting.gg/api/vouchers")
        .bearer_auth(token)
        .header("Accept", "application/json")
        .header("Content-Type", "application/json")
        .body("{\"memo\":\"Summer 2026 Promotion\",\"code\":\"SUMMER2026\",\"uses\":100,\"credits\":50,\"expires_at\":\"31-12-2026 23:59:59\"}")
        .send()
        .await?;
    let status = response.status();
    let text = response.text().await?;
    if !status.is_success() {
        return Err(format!("HTTP {}: {}", status, text).into());
    }
    println!("{}", text);
    Ok(())
}
```

### Request body (application/json)

| Property | Type | Required | Description |
| --- | --- | --- | --- |
| `memo` | string | No | Description for the voucher. |
| `code` | string | Yes | 4-36 chars, alpha-dash format. |
| `uses` | integer | Yes | Max uses. |
| `credits` | number | Yes | Credits amount. Min: 0.01, Max: 9223372036854775. |
| `expires_at` | string | No | Expiration date (d-m-Y H:i:s or d-m-Y). |
