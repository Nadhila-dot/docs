```go
package main

import (
    "fmt"
    "io"
    "net/http"
    "os"
    "time"
    "strings"
)

func main() {
    token := os.Getenv("CTRLPANEL_API_TOKEN")
    if token == "" { panic("Set CTRLPANEL_API_TOKEN first") }
    request, err := http.NewRequest("PUT", "https://ctrlpanel.hosting.gg/api/servers/X8BK1rMUV4qF3lo-ROWIl", strings.NewReader("{\"name\":\"My Awesome Server\",\"description\":\"High performance gaming server\",\"user_id\":1,\"billing_priority\":0,\"product_id\":\"1\",\"egg_id\":1,\"location_id\":1,\"egg_variables\":[]}"))
    if err != nil { panic(err) }
    request.Header.Set("Authorization", "Bearer " + token)
    request.Header.Set("Accept", "application/json")
    request.Header.Set("Content-Type", "application/json")
    client := &http.Client{Timeout: 30 * time.Second}
    response, err := client.Do(request)
    if err != nil { panic(err) }
    defer response.Body.Close()
    body, err := io.ReadAll(response.Body)
    if err != nil { panic(err) }
    if response.StatusCode < 200 || response.StatusCode >= 300 {
        panic(fmt.Sprintf("HTTP %d: %s", response.StatusCode, body))
    }
    fmt.Println(string(body))
}
```

```rust
// Dependencies:
// cargo add reqwest
// cargo add tokio --features macros,rt-multi-thread
use std::{env, time::Duration};

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    let token = env::var("CTRLPANEL_API_TOKEN")?;
    let client = reqwest::Client::builder()
        .timeout(Duration::from_secs(30))
        .build()?;
    let response = client
        .request(reqwest::Method::PUT, "https://ctrlpanel.hosting.gg/api/servers/X8BK1rMUV4qF3lo-ROWIl")
        .bearer_auth(token)
        .header("Accept", "application/json")
        .header("Content-Type", "application/json")
        .body("{\"name\":\"My Awesome Server\",\"description\":\"High performance gaming server\",\"user_id\":1,\"billing_priority\":0,\"product_id\":\"1\",\"egg_id\":1,\"location_id\":1,\"egg_variables\":[]}")
        .send()
        .await?;
    let status = response.status();
    let text = response.text().await?;
    if !status.is_success() {
        return Err(format!("HTTP {}: {}", status, text).into());
    }
    println!("{}", text);
    Ok(())
}
```

### Parameters

| Name | Location | Required | Description |
| --- | --- | --- | --- |
| `id` | path | Yes | The ID of the server. |

### Request body (application/json)

| Property | Type | Required | Description |
| --- | --- | --- | --- |
| `name` | string | No | The name of the server. |
| `description` | string | No | The description of the server. |
| `user_id` | integer | No | The ID of the user who owns the server. |
| `billing_priority` | integer | No | The billing priority. |
| `product_id` | string | No | The ID of the product. |
| `egg_id` | integer | No | The ID of the egg. |
| `location_id` | integer | No | The ID of the location. |
| `egg_variables` | object | No | The egg variables for the server. |
