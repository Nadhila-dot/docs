```go
package main

import (
    "fmt"
    "io"
    "net/http"
    "os"
    "time"
)

func main() {
    token := os.Getenv("CTRLPANEL_API_TOKEN")
    if token == "" { panic("Set CTRLPANEL_API_TOKEN first") }
    request, err := http.NewRequest("GET", "https://ctrlpanel.hosting.gg/api/vouchers/1", nil)
    if err != nil { panic(err) }
    request.Header.Set("Authorization", "Bearer " + token)
    request.Header.Set("Accept", "application/json")
    client := &http.Client{Timeout: 30 * time.Second}
    response, err := client.Do(request)
    if err != nil { panic(err) }
    defer response.Body.Close()
    body, err := io.ReadAll(response.Body)
    if err != nil { panic(err) }
    if response.StatusCode < 200 || response.StatusCode >= 300 {
        panic(fmt.Sprintf("HTTP %d: %s", response.StatusCode, body))
    }
    fmt.Println(string(body))
}
```

<Tab value="rust">

```rust
// Dependencies:
// cargo add reqwest
// cargo add tokio --features macros,rt-multi-thread
use std::{env, time::Duration};

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    let token = env::var("CTRLPANEL_API_TOKEN")?;
    let client = reqwest::Client::builder()
        .timeout(Duration::from_secs(30))
        .build()?;
    let response = client
        .request(reqwest::Method::GET, "https://ctrlpanel.hosting.gg/api/vouchers/1")
        .bearer_auth(token)
        .header("Accept", "application/json")
        .send()
        .await?;
    let status = response.status();
    let text = response.text().await?;
    if !status.is_success() {
        return Err(format!("HTTP {}: {}", status, text).into());
    }
    println!("{}", text);
    Ok(())
}
```

### Parameters

| Name | Location | Required | Description |
| --- | --- | --- | --- |
| `id` | path | Yes | The ID of the voucher. |
| `include` | query | No | Comma-separated list of related resources to include. <br />Available: <code>users</code> |
| `filter[code]` | query | No | Filter results by code |
| `filter[memo]` | query | No | Filter results by memo |
| `filter[credits]` | query | No | Filter results by credits |
| `filter[uses]` | query | No | Filter results by uses |

## Responses

### 200

| Property | Type | Required | Description |
| --- | --- | --- | --- |
| `data` | object | No |  |
