<Tab value="go">

```go
package main

import (
    "fmt"
    "io"
    "net/http"
    "os"
    "time"
    "strings"
)

func main() {
    token := os.Getenv("CTRLPANEL_API_TOKEN")
    if token == "" { panic("Set CTRLPANEL_API_TOKEN first") }
    request, err := http.NewRequest("POST", "https://ctrlpanel.hosting.gg/api/users", strings.NewReader("{\"name\":\"john_doe\",\"email\":\"john@example.com\",\"password\":\"secret123\",\"credits\":100,\"server_limit\":5,\"role_id\":1,\"referral_code\":\"ABCDEF12\"}"))
    if err != nil { panic(err) }
    request.Header.Set("Authorization", "Bearer " + token)
    request.Header.Set("Accept", "application/json")
    request.Header.Set("Content-Type", "application/json")
    client := &http.Client{Timeout: 30 * time.Second}
    response, err := client.Do(request)
    if err != nil { panic(err) }
    defer response.Body.Close()
    body, err := io.ReadAll(response.Body)
    if err != nil { panic(err) }
    if response.StatusCode < 200 || response.StatusCode >= 300 {
        panic(fmt.Sprintf("HTTP %d: %s", response.StatusCode, body))
    }
    fmt.Println(string(body))
}
```

<Tab value="rust">

```rust
// Dependencies:
// cargo add reqwest
// cargo add tokio --features macros,rt-multi-thread
use std::{env, time::Duration};

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    let token = env::var("CTRLPANEL_API_TOKEN")?;
    let client = reqwest::Client::builder()
        .timeout(Duration::from_secs(30))
        .build()?;
    let response = client
        .request(reqwest::Method::POST, "https://ctrlpanel.hosting.gg/api/users")
        .bearer_auth(token)
        .header("Accept", "application/json")
        .header("Content-Type", "application/json")
        .body("{\"name\":\"john_doe\",\"email\":\"john@example.com\",\"password\":\"secret123\",\"credits\":100,\"server_limit\":5,\"role_id\":1,\"referral_code\":\"ABCDEF12\"}")
        .send()
        .await?;
    let status = response.status();
    let text = response.text().await?;
    if !status.is_success() {
        return Err(format!("HTTP {}: {}", status, text).into());
    }
    println!("{}", text);
    Ok(())
}
```

### Request body (application/json)

| Property | Type | Required | Description |
| --- | --- | --- | --- |
| `name` | string | Yes | Min 4, max 30 chars. |
| `email` | string | Yes | Valid email address. |
| `password` | string | Yes | Min 8 chars. |
| `credits` | number | No | Initial credits. Max: 9223372036854775. |
| `server_limit` | integer | No | Initial server limit. Use 0 for unlimited. |
| `role_id` | integer | Yes | Valid role ID. |
| `referral_code` | string | No | Referral code of the user. |
